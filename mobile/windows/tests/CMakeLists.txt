cmake_minimum_required(VERSION 3.14)
project(poma_native_checks LANGUAGES CXX)
enable_testing()
add_executable(poma_native_test desktop_native_test.cpp ../runner/desktop_instance.cpp)
target_compile_features(poma_native_test PRIVATE cxx_std_17)
target_compile_definitions(poma_native_test PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_compile_options(poma_native_test PRIVATE /W4 /WX /EHsc /utf-8)
target_include_directories(poma_native_test PRIVATE ../runner)
target_link_libraries(poma_native_test PRIVATE runtimeobject shell32 advapi32 ole32 mincore)
target_link_libraries(poma_native_test PRIVATE crypt32)
target_link_libraries(poma_native_test PRIVATE winhttp)
add_test(NAME desktop_native COMMAND poma_native_test "${CMAKE_CURRENT_SOURCE_DIR}/../../test/fixtures/calendar_links.tsv")
set_tests_properties(desktop_native PROPERTIES TIMEOUT 30)
